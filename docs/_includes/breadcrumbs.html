{%- comment -%} Breadcrumbs derived from the page's source path. Each ancestor folder resolves to its index.md so the label and URL stay in sync with the content. {%- endcomment -%} {%- assign seg = page.path | split: "/" -%} {%- assign depth = seg | size -%} Home {%- if depth > 1 -%} {%- assign section_index_path = seg[0] | append: "/index.md" -%} {%- assign section_index = site.pages | where_exp: "p", "p.path == section_index_path" | first -%} {%- if section_index -%} {%- assign is_last_section = false -%} {%- if page.url == section_index.url %}{% assign is_last_section = true %}{% endif -%} {{ section_index.nav_title | default: section_index.title }} {%- endif -%} {%- endif -%} {%- if depth > 2 -%} {%- assign folder_index_path = seg[0] | append: "/" | append: seg[1] | append: "/index.md" -%} {%- assign folder_index = site.pages | where_exp: "p", "p.path == folder_index_path" | first -%} {%- if folder_index -%} {%- assign is_last_folder = false -%} {%- if page.url == folder_index.url %}{% assign is_last_folder = true %}{% endif -%} {{ folder_index.nav_title | default: folder_index.title }} {%- endif -%} {%- endif -%} {%- assign last_seg = seg | last -%} {%- if last_seg != "index.md" -%} {{ page.nav_title | default: page.title }} {%- endif -%}
